\section{Introducción}
\begin{frame}{¿Qué significó el año 70?}
\begin{columns}[t,onlytextwidth]
\begin{column}{0.50\textwidth}
\begin{cita}No quedará aquí piedra sobre piedra que no sea derribada.\fuente{(Mt 24:2)}\end{cita}
\begin{cita}¿Ha desechado Dios a su pueblo? ¡De ninguna manera! [\ldots] No ha desechado Dios a su pueblo, al cual conoció de antemano.\fuente{(Ro 11:1--2)}\end{cita}
\end{column}
\begin{column}{0.47\textwidth}\small
Dos respuestas extremas: \emph{Dios rechazó a Israel} o \emph{Jesús no tiene que ver con Israel}. Los textos no permiten ninguna.\par\medskip
\destacar{Tesis 1.} Jesús es el Mesías \emph{de Israel} y encarna su vocación; un remanente judío creyente es el núcleo del pueblo restaurado.\par\medskip
\destacar{Tesis 2.} La caída del templo es un juicio dentro del pacto, y los textos que la anuncian llevan escrito un «hasta».
\end{column}
\end{columns}
\end{frame}

\section{Historia}
\begin{frame}{Judea bajo Roma}
\begin{columns}[t,onlytextwidth]
\begin{column}{0.49\textwidth}\small
\begin{itemize}
\item Pompeyo toma Jerusalén (63 a.\,e.\,c.); Herodes reconstruye el templo (Jn 2:20).
\item Desde el 6 e.\,c., prefectos romanos en Cesarea; el sumo sacerdote preside el sanedrín.
\item Fariseos, saduceos, esenios y la «cuarta filosofía» de la resistencia (\emph{Ant.} 18.23).
\item Expectativa: «la consolación de Israel» (Lc 2:25), «la redención de Jerusalén» (2:38).
\end{itemize}
\end{column}
\begin{column}{0.49\textwidth}\small
\textbf{\textcolor{cafe}{Josefo, testigo y apologista}}
\begin{itemize}
\item Sacerdote, comandante en Galilea, se rinde en Jotapata (67) y escribe en Roma bajo los Flavios.
\item Menciona a Juan el Bautista (\emph{Ant.} 18.116--119) y a «Jacobo, hermano de Jesús, llamado Cristo» (\emph{Ant.} 20.200).
\item Sus cifras y discursos se leen con cautela (Mason, 2016).
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Precisiones a la «Octava lección»}
\small La lección «Israel, el pueblo de Dios» dice que en el 70 «más de un millón de judíos murieron», que «el sacerdocio y el sanedrín dejaron de ser» y que «comienza la Diáspora».\par\medskip
\scriptsize\renewcommand{\arraystretch}{1.25}
\begin{tabular}{>{\RaggedRight\arraybackslash}p{3.3cm}>{\RaggedRight\arraybackslash}p{8.2cm}}
\toprule
\textbf{Afirmación} & \textbf{Precisión}\\\midrule
Más de un millón de muertos & Cifra de Josefo (\emph{Guerra} 6.420), considerada exagerada.\\
Comienza la Diáspora & Ya existía desde el exilio babilónico (Hch 2:9--11). La expulsión de Jerusalén llega tras Bar Kojba (132--135): \emph{Aelia Capitolina}.\\
Fin del sacerdocio y del sanedrín & Cesan el sacrificio y el sumo sacerdocio; la tradición judía reconstituye un tribunal de sabios en Yavne.\\
Persecuciones posteriores & Expulsiones: Inglaterra 1290; Francia 1306 y 1394; España 1492. Peste negra: 1347--1351 (s. \textsc{xiv}).\\
\bottomrule
\end{tabular}\par\medskip
{\scriptsize\color{sepia}Con estos ajustes, el punto de la lección se mantiene: el juicio fue grave, pero «suspendido, mas no para siempre» (Jer 31:35--37; Ro 11:25--26).}
\end{frame}

\section{El Mesías}
\begin{frame}{«La plenitud del tiempo»}
\begin{columns}[t,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{cita}Cuando vino la plenitud del tiempo, Dios envió a su Hijo, nacido de mujer, nacido bajo la ley.\fuente{(Gá 4:4)}\end{cita}
\small \grc{τὸ πλήρωμα τοῦ χρόνου}: la palabra de la «plenitud» de Israel y de los gentiles (Ro 11:12, 25).\par\smallskip
\textbf{\textcolor{cafe}{Zacarías}} (Lc 1:68--79): Dios «ha visitado y redimido a su pueblo [\ldots] para que le sirvamos (\grc{λατρεύειν}) en santidad». La redención devuelve a Israel su vocación sacerdotal.
\end{column}
\begin{column}{0.49\textwidth}
\centering{\large\heb{קוֹל קוֹרֵא בַּמִּדְבָּר פַּנּוּ דֶּרֶךְ יְהוָה}}\par\smallskip
\begin{cita}Una voz clama: En el desierto preparad el camino de YHWH.\fuente{(Is 40:3)}\end{cita}
\small Juan el Bautista: el camino del nuevo éxodo, el consuelo de Is 40:1 y la \emph{teshuvá}.\par\smallskip
\precision{el TM une «en el desierto» con «preparad»; la LXX y Mc 1:3, con «voz». Qumrán aplicaba el texto a su retiro (1QS 8.12--16).}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Nazaret y los Doce}
\begin{columns}[t,onlytextwidth]
\begin{column}{0.49\textwidth}
\centering{\heb{לִקְרֹא לִשְׁבוּיִם דְּרוֹר}}\par\smallskip
\begin{cita}Me ha enviado a proclamar libertad a los cautivos [\ldots], a proclamar el año agradable del Señor. Hoy se ha cumplido esta Escritura.\fuente{(Lc 4:18--21; Is 61:1--2)}\end{cita}
\small \heb{דְּרוֹר} (\emph{derór}): la libertad del jubileo (Lv 25:10). Jesús se detiene antes de «el día de venganza» (Is 61:2b).
\end{column}
\begin{column}{0.49\textwidth}
\begin{cita}Os sentaréis sobre doce tronos, para juzgar a las doce tribus de Israel.\fuente{(Mt 19:28; Lc 22:30)}\end{cita}
\scriptsize
\begin{itemize}
\item Doce: signo de la reconstitución de \emph{todo} Israel, incluidas las tribus del norte (Ez 37:15--28; Sanders, 1985).
\item Matías completa el número (Hch 1:15--26); «nuestras doce tribus» (Hch 26:7).
\item «Las ovejas perdidas de la casa de Israel» (Mt 10:6; 15:24; Jer 50:6; Ez 34).
\end{itemize}
\precision{en Mateo, «casa de Israel» es el pueblo judío de Galilea y Judea, no el antiguo reino del norte.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Jesús en las fiestas de Israel}
\begin{columns}[t,onlytextwidth]

\textbf{\textcolor{ladrillo}{Pésaj y el Hallel}}\par\smallskip

\end{column}
\end{columns}\par\medskip

\begin{frame}{Los labradores: Mt 21:43}
\begin{columns}[t,onlytextwidth]
\begin{column}{0.44\textwidth}
\begin{cita}El reino de Dios será quitado de vosotros y será dado a un pueblo que produzca sus frutos.\fuente{(Mt 21:43)}\end{cita}
\small Antes, en el templo: «casa de oración [\ldots] para todos los pueblos» (Is 56:7) y «cueva de ladrones» (Jer 7:11), el sermón que anunció la caída del Primer Templo.
\end{column}
\begin{column}{0.53\textwidth}\scriptsize
\textbf{\textcolor{cafe}{¿Reemplazo de Israel? El texto dice otra cosa}}
\begin{enumerate}
\item \textbf{Destinatarios}: los principales sacerdotes y los fariseos «entendieron que hablaba de ellos» (21:45).
\item \textbf{La viña queda}: es Israel (Is 5:7); se entrega «a otros labradores» (21:41).
\item \textbf{Singular}: \grc{ἔθνει}, «a un pueblo», no \grc{τοῖς ἔθνεσιν}, «a los gentiles». Su núcleo: los Doce (19:28).
\item \textbf{La piedra desechada} (Sal 118:22) es el Hijo; Pedro la aplica a los «edificadores» (Hch 4:11).
\end{enumerate}
\destacar{Cambio de liderazgo, no de pueblo} (Snodgrass, 2008).
\end{column}
\end{columns}
\end{frame}

\section{Reflexión}
\begin{frame}{Para la reflexión}
\small
\begin{enumerate}
\item Simeón une «luz para las naciones» y «gloria de tu pueblo Israel» (Lc 2:32). ¿Qué se pierde cuando se conserva solo la primera mitad?
\item ¿Por qué importa que Mt 21:43 hable de los dirigentes y de «un pueblo» en singular?
\item Jesús llora sobre Jerusalén antes de anunciar su ruina. ¿Qué actitud pide esto a los creyentes de las naciones (Ro 11:18--21)?
\item Yohanán ben Zakai y Jesús citaron Os 6:6. ¿En qué se parecen y en qué se diferencian sus respuestas a la pérdida del templo?
\item ¿Qué «hasta» de esta etapa le parece más significativo para la esperanza de Israel?
\end{enumerate}
\end{frame}
